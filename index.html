function runJarvis() { let text = document.getElementById("command").value.toLowerCase().trim(); let reply = ""; if (text === "") { reply = "Please type a command."; } else if (text.includes("hello") || text.includes("hi")) { reply = "Hello! I am JARVIS. How can I help you?"; } else if (text.includes("your name")) { reply = "My name is JARVIS, your personal assistant."; } else if (text.includes("time")) { reply = "The current time is " + new Date().toLocaleTimeString(); } else if (text.includes("date")) { reply = "Today's date is " + new Date().toLocaleDateString(); } else if (text.includes("how are you")) { reply = "I am functioning perfectly. Thank you!"; } else if (text.includes("who created you")) { reply = "I was created as your personal JARVIS project."; } else if (text.includes("thank you")) { reply = "You are welcome!"; } else if (text.includes("help")) { reply = "You can ask my name, time, date, or say hello."; } else { reply = "I don't know that command yet. Please try another command."; } document.getElementById("response").innerText = reply; speak(reply); }
